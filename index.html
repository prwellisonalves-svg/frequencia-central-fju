SQL para criar tabela e bucket no Supabase 
-- 1. Tabela de frequência das aulas da Central FJU
create table if not exists public.frequencia_aulas (
  id uuid primary key default gen_random_uuid(),
  class_id text not null,
  class_name text not null,
  day_code text not null,
  scheduled_time text not null,
  lesson_date date not null,
  total_students integer not null check (total_students >= 0),
  first_time_students integer not null check (first_time_students >= 0 and first_time_students <= total_students),
  theme text not null,
  teacher text not null,
  photos jsonb not null default '[]'::jsonb,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- 2. Habilitar RLS e políticas públicas para acesso via link
alter table public.frequencia_aulas enable row level security;

create policy "Acesso publico frequencia_aulas"
on public.frequencia_aulas
for all
using (true)
with check (true);

-- 3. Bucket público para fotos das aulas
insert into storage.buckets (id, name, public)
values ('aula-fotos', 'aula-fotos', true)
on conflict (id) do nothing;

create policy "Leitura publica aula-fotos"
on storage.objects for select
using (bucket_id = 'aula-fotos');

create policy "Upload publico aula-fotos"
on storage.objects for insert
with check (bucket_id = 'aula-fotos');

create policy "Exclusao publica aula-fotos"
on storage.objects for delete
using (bucket_id = 'aula-fotos');
